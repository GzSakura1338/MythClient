set(WEBVIEW2_SDK "${CMAKE_CURRENT_SOURCE_DIR}/vendor/webview2-sdk" CACHE PATH "Pinned WebView2 SDK")

option(NOVA_CHROME_HOST "Build the bundled WebView2 host" ON)
if(NOVA_CHROME_HOST)
    if(NOT EXISTS "${WEBVIEW2_SDK}/build/native/include/WebView2.h" OR NOT EXISTS "${WEBVIEW2_RUNTIME}/msedgewebview2.exe")
        message(FATAL_ERROR "Run Client/src/chrome/tools/get-webview2.ps1 before building the browser host.")
    endif()
    add_executable(NovaBrowserHost WIN32 host/WebViewHost.cpp)
    target_include_directories(NovaBrowserHost PRIVATE "${WEBVIEW2_SDK}/build/native/include" "${NOVA_CHROME_CLIENT_ROOT}/src")
    target_compile_features(NovaBrowserHost PRIVATE cxx_std_20)
    target_compile_definitions(NovaBrowserHost PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_compile_options(NovaBrowserHost PRIVATE /utf-8 /EHsc /bigobj)
    target_link_libraries(NovaBrowserHost PRIVATE "${WEBVIEW2_SDK}/build/native/x64/WebView2LoaderStatic.lib" user32 ole32 oleaut32 shell32 shlwapi version d3d11 dxgi windowsapp CoreMessaging)
    set_target_properties(NovaBrowserHost PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/browser-webview2")
    add_custom_command(TARGET NovaBrowserHost POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_directory "${WEBVIEW2_RUNTIME}" "$<TARGET_FILE_DIR:NovaBrowserHost>/runtime")
endif()
